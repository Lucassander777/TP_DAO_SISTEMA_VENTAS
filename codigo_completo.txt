from abc import ABC, abstractmethod
from domain.exceptions import TransicionEstadoInvalidaException

class EstadoBulto(ABC):
    @abstractmethod
    def precintar(self, bulto):
        pass

    @abstractmethod
    def despachar(self, bulto):
        pass

    @abstractmethod
    def recibir(self, bulto):
        pass

    @abstractmethod
    def verificar(self, bulto):
        pass

    @abstractmethod
    def rechazar(self, bulto):
        pass

class Preparado(EstadoBulto):
    def precintar(self, bulto):
        bulto.estado_actual = Precintado()

    def despachar(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto debe estar precintado para despacharse.")

    def recibir(self, bulto):
        raise TransicionEstadoInvalidaException("No se puede recibir un bulto recién preparado.")

    def verificar(self, bulto):
        raise TransicionEstadoInvalidaException("No se puede verificar un bulto recién preparado.")

    def rechazar(self, bulto):
        raise TransicionEstadoInvalidaException("No se puede rechazar un bulto recién preparado.")

class Precintado(EstadoBulto):
    def precintar(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto ya está precintado.")
    
    def despachar(self, bulto):
        bulto.estado_actual = EnTransito()
    
    def recibir(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto debe despacharse antes de recibirse.")
    
    def verificar(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto debe recibirse antes de verificarse.")
    
    def rechazar(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto debe recibirse antes de rechazarse.")

class EnTransito(EstadoBulto):
    def precintar(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto ya está en tránsito.")
    
    def despachar(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto ya fue despachado.")
    
    def recibir(self, bulto):
        bulto.estado_actual = Recibido()
    
    def verificar(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto debe recibirse antes de verificarse.")
    
    def rechazar(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto debe recibirse antes de rechazarse.")

class Recibido(EstadoBulto):
    def precintar(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto ya fue recibido.")
    
    def despachar(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto ya fue recibido.")
    
    def recibir(self, bulto):
        raise TransicionEstadoInvalidaException("El bulto ya fue recibido.")
    
    def verificar(self, bulto):
        bulto.estado_actual = Verificado()

from abc import ABC, abstractmethod
from decimal import Decimal

class MedioPago(ABC):
    @abstractmethod
    def validar(self, monto: Decimal):
        pass
        
    @abstractmethod
    def procesar_pago(self, monto: Decimal):
        pass

class PagoEfectivo(MedioPago):
    def validar(self, monto: Decimal):
        # El efectivo no requiere validación extra de comprobantes
        return True
        
    def procesar_pago(self, monto: Decimal):
        pass

    def __init__(self, numero_tarjeta: str, cuotas: int):

    def validar(self, monto: Decimal):
        if not self.numero_tarjeta:
            raise ValueError("Falta el número de tarjeta")
        return True

    def procesar_pago(self, monto: Decimal):
        pass

class PagoTransferencia(MedioPago):

class Promocion(ABC):
    def __init__(self, nombre: str, descripcion: str, vigente_desde: date, vigente_hasta: date, activa: bool):
        self.nombre = nombre
        self.descripcion = descripcion
        self.vigente_desde = vigente_desde
        self.vigente_hasta = vigente_hasta
        self.activa = activa

    @abstractmethod
    def aplica(self, carrito_items: list) -> bool:
        pass

    @abstractmethod
    def calcular_descuento(self, carrito_items: list) -> Decimal:
        pass

class DescuentoPorcentaje(Promocion):
    def __init__(self, nombre: str, descripcion: str, vigente_desde: date, vigente_hasta: date, activa: bool, porcentaje: Decimal):
        super().__init__(nombre, descripcion, vigente_desde, vigente_hasta, activa)
        self.porcentaje = porcentaje

    def aplica(self, carrito_items: list) -> bool:
        hoy = date.today()
        return self.activa and (self.vigente_desde <= hoy <= self.vigente_hasta)
    
    def calcular_descuento(self, carrito_items: list) -> Decimal:
        if not self.aplica(carrito_items):
            return Decimal("0.00")
        
        total_carrito = sum(item['precio'] * item['cantidad'] for item in carrito_items)
        return total_carrito * (self.porcentaje / Decimal("100"))

class DescuentoMontoFijo(Promocion):
    def __init__(self, nombre: str, descripcion: str, vigente_desde: date, vigente_hasta: date, activa: bool, monto: Decimal):
        super().__init__(nombre, descripcion, vigente_desde, vigente_hasta, activa)
        self.monto = monto

    def aplica(self, carrito_items: list) -> bool:
        hoy = date.today()
        return self.activa and (self.vigente_desde <= hoy <= self.vigente_hasta)
    
    def calcular_descuento(self, carrito_items: list) -> Decimal:
        if not self.aplica(carrito_items):
            return Decimal("0.00")
            
        total_carrito = sum(item['precio'] * item['cantidad'] for item in carrito_items)
        return min(self.monto, total_carrito)

class PromoNxM(Promocion):
    def __init__(self, nombre: str, descripcion: str, vigente_desde: date, vigente_hasta: date, activa: bool, lleva: int, paga: int):
        super().__init__(nombre, descripcion, vigente_desde, vigente_hasta, activa)
        self.lleva = lleva
        self.paga = paga

    def aplica(self, carrito_items: list) -> bool:
        hoy = date.today()
        if not (self.activa and (self.vigente_desde <= hoy <= self.vigente_hasta)):
            return False
        # Aplica si al menos un producto en el carrito alcanza la cantidad 'lleva'
        return any(item['cantidad'] >= self.lleva for item in carrito_items)
    
    def calcular_descuento(self, carrito_items: list) -> Decimal:
        if not self.aplica(carrito_items):
            return Decimal("0.00")
            
        descuento_total = Decimal("0.00")
        for item in carrito_items:
            if item['cantidad'] >= self.lleva:
                grupos = item['cantidad'] // self.lleva
                unidades_gratis = grupos * (self.lleva - self.paga)
                descuento_total += item['precio'] * Decimal(unidades_gratis)
                
        return descuento_total

        cursor = conexion.cursor()

            SELECT s.cantidad_fisica, COALESCE(p.nombre, 'Producto sin registrar')
            FROM Stock s
            LEFT JOIN Productos p ON s.id_producto = p.id
            WHERE s.id_producto = ? AND s.id_ubicacion = ?
        """, (id_producto, id_ubicacion))

        resultado = cursor.fetchone()
        conexion.close()
        return resultado

    def ajustar_stock(self, id_producto, id_ubicacion, nueva_cantidad, diferencia):
        conexion = self.get_conexion()
        cursor = conexion.cursor()

        # 1. Verificamos si ya existe un registro de stock para este producto y ubicación
        cursor.execute("SELECT 1 FROM Stock WHERE id_producto = ? AND id_ubicacion = ?", (id_producto, id_ubicacion))
        existe = cursor.fetchone()

        if existe:
            # Si existe, actualizamos
            cursor.execute("""
                UPDATE Stock
                SET cantidad_fisica = ?, cantidad_disponible = ?
                WHERE id_producto = ? AND id_ubicacion = ?
            """, (nueva_cantidad, nueva_cantidad, id_producto, id_ubicacion))
        else:
            # Si NO existe, lo insertamos para evitar errores de integridad o claves foráneas
            cursor.execute("""
                INSERT INTO Stock (id_producto, id_ubicacion, cantidad_fisica, cantidad_disponible)
                VALUES (?, ?, ?, ?)
            """, (id_producto, id_ubicacion, nueva_cantidad, nueva_cantidad))

        # 2. Registramos el movimiento de auditoría con la fecha formateada en texto
        cursor.execute("""

        """, (

            diferencia,
            'AJUSTE_AUDITORIA',
            'Conciliacion Fisica',
            0,

class ComprasDAO(BaseDAO):
    def __init__(self):
        super().__init__()

    def obtener_faltantes_stock(self):
        conexion = self.get_conexion()
        cursor = conexion.cursor()

        cursor.execute("""
            SELECT 
                p.id, 
                p.nombre, 
                COALESCE(SUM(s.cantidad_disponible), 0) as stock_real, 
                p.stock_minimo
            FROM Productos p
            LEFT JOIN Stock s ON p.id = s.id_producto
            WHERE p.activo = 1
            GROUP BY p.id, p.nombre, p.stock_minimo
            HAVING stock_real < p.stock_minimo
        """)

        faltantes = cursor.fetchall()
        conexion.close()

        lista_faltantes = []
        for f in faltantes:
            lista_faltantes.append({
                'producto_id': f[0],
                'producto_nombre': f[1],
                'stock_actual': f[2],
                'stock_minimo': f[3],
                'cantidad_sugerida': f[3] - f[2]
            })
        return lista_faltantes

    def guardar_orden(self, orden: OrdenCompra) -> OrdenCompra:
        conexion = self.get_conexion()

import sqlite3

class Database:
    def __init__(self, db_name="sistema_ventas.db"):
        self.db_name = db_name

    def get_conexion(self):
        conexion = sqlite3.connect(self.db_name)
        conexion.row_factory = sqlite3.Row # Mejor usar sqlite3.Row para acceder por nombre de columna

        conexion.execute("PRAGMA foreign_keys = ON")
        return conexion

    def incializar_tablas(self):
        """Crea todo el esquema relacional de la base de datos"""
        conexion = self.get_conexion()
        cursor = conexion.cursor()

        # ----------
        # TABLAS MAESTRAS (sin dependencias)
        # ----------

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Proveedores (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                razon_social TEXT NOT NULL,
                cuit TEXT UNIQUE NOT NULL
            )
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Categorias(
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                nombre TEXT NOT NULL
                )
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Ubicaciones(
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                nombre TEXT NOT NULL,
                direccion TEXT,
                tipo TEXT NOT NULL -- Ej: 'Local', 'CentroDistribucion'
            )
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Usuarios(
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT UNIQUE NOT NULL,
                rol TEXT NOT NULL
            )
        """)

        # ----------
        # CATÁLOGO Y LOTES (dependen de las maestras)
        # ----------

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Productos(
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                sku TEXT UNIQUE NOT NULL,
                nombre TEXT NOT NULL,
                precio_venta REAL NOT NULL,
                costo REAL NOT NULL,
                stock_actual INTEGER DEFAULT 0, 
                stock_minimo INTEGER DEFAULT 5,
                activo BOOLEAN NOT NULL DEFAULT 1,
                id_categoria INTEGER,
                id_proveedor INTEGER,
                FOREIGN KEY (id_categoria) REFERENCES Categorias(id),
                FOREIGN KEY (id_proveedor) REFERENCES Proveedores(id)
            )
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Lotes(
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                codigo TEXT UNIQUE NOT NULL,
                fecha_ingreso DATE NOT NULL,
                fecha_vencimiento DATE NOT NULL,
                activo BOOLEAN NOT NULL DEFAULT 1,
                id_producto INTEGER NOT NULL,
                id_proveedor INTEGER,
                FOREIGN KEY (id_producto) REFERENCES Productos(id),
                FOREIGN KEY (id_proveedor) REFERENCES Proveedores(id)
            )
        """)
        # ----------
        # INVENTARIO (depende de productos y ubicaciones)
        # ----------
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Stock(
                id_producto INTEGER NOT NULL,
                id_ubicacion INTEGER NOT NULL,
                cantidad_fisica INTEGER NOT NULL DEFAULT 0,
                cantidad_disponible INTEGER NOT NULL DEFAULT 0,
                PRIMARY KEY (id_producto, id_ubicacion),
                FOREIGN KEY (id_producto) REFERENCES Productos(id),
                FOREIGN KEY (id_ubicacion) REFERENCES Ubicaciones(id)
            )
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS MovimientoStock(
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                fecha DATETIME NOT NULL,
                cantidad INTEGER NOT NULL,
                tipo TEXT NOT NULL, -- Ej: 'VENTA', 'RECEPCION'
                referencia_tipo TEXT,
                referencia_id INTEGER,
                id_producto INTEGER NOT NULL,
                id_ubicacion INTEGER NOT NULL,
                FOREIGN KEY (id_producto) REFERENCES Productos(id),
                FOREIGN KEY (id_ubicacion) REFERENCES Ubicaciones(id)
            )
        """)

        # ----------
        # LOGÍSTICA (Bultos y trazabilidad)
        # ----------

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Bultos (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                codigo TEXT UNIQUE NOT NULL,
                fecha_creacion DATETIME NOT NULL,
                estado_actual TEXT NOT NULL,
                id_origen INTEGER,
                id_destino INTEGER,
                FOREIGN KEY (id_origen) REFERENCES Ubicaciones(id),
                FOREIGN KEY (id_destino) REFERENCES Ubicaciones(id)
            )
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS ItemBulto(
                id_bulto INTEGER NOT NULL,
                id_producto INTEGER NOT NULL,
                cantidad INTEGER NOT NULL,
                PRIMARY KEY (id_bulto, id_producto),
                FOREIGN KEY (id_bulto) REFERENCES Bultos(id),
                FOREIGN KEY (id_producto) REFERENCES Productos(id)
            )
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS CambioEstadoBulto(
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                id_bulto INTEGER NOT NULL,
                estado TEXT NOT NULL,
                fecha_hora DATETIME NOT NULL,
                usuario TEXT NOT NULL,
                observacion TEXT,
                FOREIGN KEY (id_bulto) REFERENCES Bultos(id)
            )
        """)

        # ----------
        # VENTAS (Cabecera, detalle, pagos y devoluciones)
        # ----------

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Ventas(
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                fecha DATETIME NOT NULL,
                estado TEXT NOT NULL,
                total REAL NOT NULL,
                id_cajero INTEGER,
                id_local INTEGER,
                FOREIGN KEY (id_cajero) REFERENCES Usuarios(id),
                FOREIGN KEY (id_local) REFERENCES Ubicaciones(id)
            )
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS ItemVenta(
                id_venta INTEGER NOT NULL,
                id_producto INTEGER NOT NULL,
                cantidad INTEGER NOT NULL,
                precio_unitario REAL NOT NULL,
                PRIMARY KEY (id_venta, id_producto),
                FOREIGN KEY (id_venta) REFERENCES Ventas(id),
                FOREIGN KEY (id_producto) REFERENCES Productos(id)
            )
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Pagos(
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                id_venta INTEGER NOT NULL,
                monto REAL NOT NULL,
                referencia TEXT,
                tipo_medio_pago TEXT NOT NULL,
                FOREIGN KEY (id_venta) REFERENCES Ventas(id)
            )
        """)

        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Comprobantes (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                id_venta INTEGER UNIQUE NOT NULL,
                tipo TEXT NOT NULL,
                numero TEXT NOT NULL,
                fecha DATETIME NOT NULL,
                FOREIGN KEY (id_venta) REFERENCES Ventas(id)
            )
        """)

        # NUEVA TABLA: Devoluciones
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS Devoluciones (
                id INTEGER PRIMARY KEY AUTOINCREMENT,

    def actualizar_stock(self, id_producto: int, id_ubicacion: int, cantidad_fisica: int, cantidad_disponible: int):
        """Inserta el registro de stock si no existe, o lo actualiza si ya existe"""
        conexion = self.get_conexion()
        cursor = conexion.cursor()

        cursor.execute("SELECT 1 FROM Stock WHERE id_producto=? AND id_ubicacion=?", (id_producto, id_ubicacion))
        existe = cursor.fetchone()

        if existe:
            sql = """UPDATE Stock SET cantidad_fisica=?, cantidad_disponible=? WHERE id_producto=? AND id_ubicacion=?"""
            cursor.execute(sql, (cantidad_fisica, cantidad_disponible, id_producto, id_ubicacion))
        else:
            sql = """INSERT INTO Stock (id_producto, id_ubicacion, cantidad_fisica, cantidad_disponible) VALUES (?, ?, ?, ?)"""
            cursor.execute(sql, (id_producto, id_ubicacion, cantidad_fisica, cantidad_disponible))
        conexion.commit()
        conexion.close()

    def registrar_movimiento(self, movimiento: MovimientoStock, id_producto: int, id_ubicacion: int):

        sql = """INSERT INTO MovimientoStock (fecha, cantidad, tipo, referencia_tipo, referencia_id, id_producto, id_ubicacion) VALUES (?, ?, ?, ?, ?, ?, ?)"""

        cursor.execute(sql, (

    def consultar_stock_disponible(self, id_producto: int, id_ubicacion: int) -> int:

        conexion = self.get_conexion()
        cursor = conexion.cursor()
        
        sql = "SELECT cantidad_disponible FROM Stock WHERE id_producto = ? AND id_ubicacion = ?"
        cursor.execute(sql, (id_producto, id_ubicacion))
        fila = cursor.fetchone()
        conexion.close()
        
        if fila:
            return fila[0]
        return 0

    def obtener_faltantes_stock(self):
        """Busca productos con stock disponible inferior al stock mínimo"""
        conexion = self.get_conexion()
        cursor = conexion.cursor()

        cursor.execute("PRAGMA table_info(Productos)")
        columnas = [col[1] for col in cursor.fetchall()]
        col_minimo = "p.stock_minimo" if "stock_minimo" in columnas else "10"

        sql = f"""
            SELECT 
                p.nombre AS producto_nombre,
                COALESCE(SUM(s.cantidad_disponible), 0) AS stock_actual,
                {col_minimo} AS stock_minimo
            FROM Productos p
            LEFT JOIN Stock s ON p.id = s.id_producto
            GROUP BY p.id, p.nombre
            HAVING stock_actual < stock_minimo
        """
        cursor.execute(sql)
        filas = cursor.fetchall()
        conexion.close()

        faltantes = []
        for fila in filas:
            nombre = fila['producto_nombre'] if hasattr(fila, '__getitem__') and isinstance(fila, dict) or hasattr(fila, 'keys') else fila[0]
            actual = fila['stock_actual'] if hasattr(fila, '__getitem__') and isinstance(fila, dict) or hasattr(fila, 'keys') else fila[1]
            minimo = fila['stock_minimo'] if hasattr(fila, '__getitem__') and isinstance(fila, dict) or hasattr(fila, 'keys') else fila[2]

            sugerido = max(0, minimo - actual)
            faltantes.append({
                'producto_nombre': nombre,
                'stock_actual': actual,
                'stock_minimo': minimo,
                'cantidad_sugerida': sugerido
            })
        return faltantes

    def obtener_inventario_completo(self):
        """Trae todo el stock cruzando con Productos y Ubicaciones para la grilla visual"""
        conexion = self.get_conexion()
        cursor = conexion.cursor()
        
        sql = """
            SELECT 
                COALESCE(p.nombre, 'Prod ' || s.id_producto) as producto,
                COALESCE(u.nombre, 'Ubic ' || s.id_ubicacion) as ubicacion,
                s.cantidad_fisica,
                s.cantidad_disponible
            FROM Stock s
            LEFT JOIN Productos p ON s.id_producto = p.id
            LEFT JOIN Ubicaciones u ON s.id_ubicacion = u.id
        """
        cursor.execute(sql)
        filas = cursor.fetchall()
        conexion.close()
        
        
        return [dict(fila) for fila in filas]

class AuditoriaService:

            raise NegocioException("La cantidad física no puede ser negativa.")

        datos_sistema = self.dao.obtener_stock_actual(id_producto, id_ubicacion)
        if not datos_sistema:
            raise NegocioException(f"No hay registro de stock para el producto {id_producto} en la ubicación {id_ubicacion}.")

        nombre_producto = datos_sistema[1]

        diferencia = cantidad_fisica - stock_sistema

        self.dao.ajustar_stock(id_producto, id_ubicacion, cantidad_fisica, diferencia)

        if diferencia < 0:
            return f"Merma detectada: faltan {abs(diferencia)} unidades de {nombre_producto}."
        elif diferencia > 0:
            return f"Sobrante detectado: sobran {diferencia} unidades de {nombre_producto}."
        else:
            return f"Conciliación perfecta. El stock de {nombre_producto} coincide."

        )

            producto = data['item_orden'].producto

            nuevo_lote = Lote(
                id_lote=None,
                codigo=f"LOTE-{producto.sku}-{datetime.now().strftime('%Y%m%d')}",

                producto=producto,
                proveedor=orden_compra.proveedor
            )

                id_orden=data['item_orden'],
                cantidad=data['cantidad'],
                costo_unitario=data['costo'],
                lote=nuevo_lote
            )
            recepcion.items.append(item_recepcion)

            # Sumar al inventario
            if self.inventario_service:
                # Se asume que destino es Ubicacion 1 (Centro Distribucion). Ajustá si es dinámico.
                id_destino = orden_compra.destino.id if hasattr(orden_compra, 'destino') and orden_compra.destino else 1

                    id_producto=producto.id,
                    id_ubicacion=id_destino,
                    cantidad=data['cantidad'],
                    tipo=TipoMovimiento.RECEPCION,
                    referencia_tipo="RecepcionCompra",
                    referencia_id=orden_compra.id if hasattr(orden_compra, 'id') else 0

        if hasattr(orden_compra, 'id') and orden_compra.id:

from datetime import datetime
from domain.models.inventario import MovimientoStock, TipoMovimiento
from domain.exceptions import StockInsuficienteException
from repositories.inventario_dao import InventarioDao

class InventarioService:
    def __init__(self, inventario_dao: InventarioDao):
        # Guardamos ambas referencias para que NUNCA más falle por el nombre del atributo
        self.inventario_dao = inventario_dao
        self.dao = inventario_dao  

    def obtener_faltantes(self) -> list:
        return self.inventario_dao.obtener_faltantes_stock()

    def consultar_stock_disponible(self, id_producto: int, id_ubicacion: int) -> int:
        stock = self.inventario_dao.obtener_stock(id_producto, id_ubicacion)
        return stock.cantidad_disponible if stock else 0

    def registrar_entrada(self, id_producto: int, id_ubicacion: int, cantidad: int, tipo: TipoMovimiento, referencia_tipo: str, referencia_id: int):
        if cantidad <= 0:
            raise ValueError("La cantidad de entrada debe ser mayor a cero.")

        stock = self.inventario_dao.obtener_stock(id_producto, id_ubicacion)
        cantidad_fisica = (stock.cantidad_fisica if stock else 0) + cantidad
        cantidad_disponible = (stock.cantidad_disponible if stock else 0) + cantidad

        self.inventario_dao.actualizar_stock(id_producto, id_ubicacion, cantidad_fisica, cantidad_disponible)

        movimiento = MovimientoStock(
            id_movimiento=None,
            fecha=datetime.now(),
            cantidad=cantidad,
            tipo=tipo,
            referencia_tipo=referencia_tipo,
            referencia_id=referencia_id,
            producto=None,
            ubicacion=None,
            usuario=None
        )
        self.inventario_dao.registrar_movimiento(movimiento, id_producto, id_ubicacion)
        return movimiento

    def registrar_salida(self, id_producto: int, id_ubicacion: int, cantidad: int, tipo: TipoMovimiento, referencia_tipo: str, referencia_id: int):
        if cantidad <= 0:
            raise ValueError("La cantidad de salida debe ser mayor a cero.")

        stock_actual = self.inventario_dao.consultar_stock_disponible(id_producto, id_ubicacion)
        if stock_actual < cantidad:
            raise StockInsuficienteException(f"No hay stock suficiente. Requerido: {cantidad}, Disponible: {stock_actual}")

        stock = self.inventario_dao.obtener_stock(id_producto, id_ubicacion)
        nueva_fisica = stock.cantidad_fisica - cantidad
        nueva_disponible = stock.cantidad_disponible - cantidad

        self.inventario_dao.actualizar_stock(id_producto, id_ubicacion, nueva_fisica, nueva_disponible)

        movimiento = MovimientoStock(
            id_movimiento=None,
            fecha=datetime.now(),
            cantidad=-cantidad,
            tipo=tipo,
            referencia_tipo=referencia_tipo,
            referencia_id=referencia_id,
            producto=None,
            ubicacion=None,
            usuario=None
        )
        self.inventario_dao.registrar_movimiento(movimiento, id_producto, id_ubicacion)
        return movimiento

    def obtener_inventario_completo(self):
        return self.inventario_dao.obtener_inventario_completo()

from datetime import datetime
from domain.models.logistica import Bulto, ItemBulto, CambioEstadoBulto

class LogisticaService:
    def __init__(self, bulto_dao: BultoDAO, inventario_service: InventarioService = None):
        self.dao = bulto_dao
        self.inventario_service = inventario_service

    def armar_bulto(self, codigo: str, origen, destino, items_dict: dict, usuario: str) -> Bulto:
        """Crea un nuevo bulto desde cero. items_dict es un diccionario de {producto: cantidad}."""
        nuevo_bulto = Bulto(
            id_bulto=None,
            codigo=codigo,
            fecha_creacion=datetime.now(),
            origen=origen,
            destino=destino
        )
        nuevo_bulto.estado_actual = Preparado()
        nuevo_bulto = self.dao.guardar(nuevo_bulto)

        for producto, cantidad in items_dict.items():
            item = ItemBulto(
                producto=producto,
                cantidad=cantidad
            )
            self.dao.agregar_item(nuevo_bulto.id, item)

        cambio_inicial = CambioEstadoBulto(
            id_cambio=None,
            estado="Preparado",
            fecha_hora=datetime.now(),
            usuario=usuario,
            observacion="Armado inicial del bulto"
        )
        self.dao.registrar_cambio_estado(nuevo_bulto.id, cambio_inicial)
        return nuevo_bulto

    def precintar_bulto(self, id_bulto: int, usuario: str, codigo_precinto: str):
        bulto = self.dao.obtener_por_id(id_bulto)
        if not bulto:
            raise ValueError("El bulto no existe.")
        
        bulto.estado_actual.precintar(bulto)
        self.dao.guardar(bulto)

        cambio = CambioEstadoBulto(
            id_cambio=None,
            estado="Precintado",
            fecha_hora=datetime.now(),
            usuario=usuario,
            observacion=f'Precinto colocado: {codigo_precinto}'
        )
        self.dao.registrar_cambio_estado(bulto.id, cambio)

    def despachar_bulto(self, id_bulto: int, usuario: str, nro_remito: str = None):
        bulto = self.dao.obtener_por_id(id_bulto)
        bulto.estado_actual.despachar(bulto)
        self.dao.guardar(bulto)

        observacion = f"Despachado con remito {nro_remito}" if nro_remito else "Despachado sin remito"
        cambio = CambioEstadoBulto(
            id_cambio=None,
            estado="EnTransito",
            fecha_hora=datetime.now(),
            usuario=usuario,
            observacion=observacion
        )
        self.dao.registrar_cambio_estado(bulto.id, cambio)

    def actualizar_estado_bulto(self, codigo: str, estado_str: str, usuario: str, observacion: str):
        """Busca un bulto por su código y le cambia el estado, registrando la auditoría."""
        bulto = self.dao.obtener_por_codigo(codigo)
        if not bulto:
            raise ValueError(f"No se encontró ningún bulto con el código '{codigo}'")

        # 1. Registramos el cambio de estado en la cabecera e historial
        cambio = CambioEstadoBulto(
            id_cambio=None,
            estado=estado_str,
            fecha_hora=datetime.now(),
            usuario=usuario,
            observacion=observacion
        )
        self.dao.registrar_cambio_estado(bulto.id, cambio)
        self.dao.actualizar_estado_cabecera(bulto.id, estado_str)

        # 2. REGLA DE NEGOCIO LOGÍSTICA: 
        # Si se verifica la entrega, restamos stock del origen y sumamos en destino
        if estado_str == "Verificado" and self.inventario_service:
            items_bulto = self.dao.obtener_items_por_bulto(bulto.id)
            
            for item in items_bulto:
                id_prod = item.producto.id
                cant = item.cantidad
                
                # A) RESTAR DEL ORIGEN (Centro de Distribución)
                if bulto.origen:
                    stock_origen = self.inventario_service.dao.obtener_stock(id_prod, bulto.origen.id)
                    if stock_origen:
                        nueva_fisica_orig = stock_origen.cantidad_fisica - cant
                        nueva_disp_orig = stock_origen.cantidad_disponible - cant
                        
                        # Control de seguridad por si despachan más de lo que hay
                        if nueva_fisica_orig < 0: nueva_fisica_orig = 0
                        if nueva_disp_orig < 0: nueva_disp_orig = 0
                        
                        self.inventario_service.dao.actualizar_stock(
                            id_prod, bulto.origen.id, nueva_fisica_orig, nueva_disp_orig
                        )

                # B) SUMAR AL DESTINO (Local)
                if bulto.destino:
                    stock_destino = self.inventario_service.dao.obtener_stock(id_prod, bulto.destino.id)
                    if stock_destino:
                        nueva_fisica_dest = stock_destino.cantidad_fisica + cant
                        nueva_disp_dest = stock_destino.cantidad_disponible + cant
                        self.inventario_service.dao.actualizar_stock(
                            id_prod, bulto.destino.id, nueva_fisica_dest, nueva_disp_dest
                        )
                    else:
                        # Si el local nunca tuvo este producto, se crea la fila de stock de cero
                        self.inventario_service.dao.actualizar_stock(
                            id_prod, bulto.destino.id, cant, cant
                        )

        return bulto

from datetime import datetime
from decimal import Decimal
from domain.models.ventas import Venta, ItemVenta, Pago, Comprobante
from domain.models.pagos_strategy import MedioPago
from domain.models.inventario import TipoMovimiento
from domain.exceptions import NegocioException
from repositories.venta_dao import VentaDAO
from services.inventario_service import InventarioService

class VentaService:
    def __init__(self, venta_dao: VentaDAO, inventario_service: InventarioService):
        self.dao = venta_dao
        self.inventario_service = inventario_service

    def procesar_venta_completa(self, cajero, local, carrito_items: list, medio_pago: MedioPago, promocion=None) -> Venta:
        """Orquesta todo el flujo de una venta: creación, descuento de stock, cobro y facturación. carrito_items es una lista de diccionarios con {'producto': prod, 'cantidad': cant, 'precio': precio}"""
        if not carrito_items:
            raise NegocioException("No se puede procesaro una venta sin productos en el carrito")

        # 1- Calculamos el total bruto inical
        total_bruto = sum(item['cantidad'] * item['precio'] for item in carrito_items)
        total_venta = Decimal(str(total_bruto))

        if promocion is not None and promocion.aplica(carrito_items):
            descuento = promocion.calcular_descuento(carrito_items)
            total_venta -= descuento

        nueva_venta = Venta(
            id_venta=None,
            fecha=datetime.now(),
            estado="Completada",
            total=total_venta,
            cajero=cajero,
            local=local
        )
        # Guardamos la cabecera en SQLite
        nueva_venta = self.dao.guardar(nueva_venta)

        # 2- Guardamos el detalle y descontamos el stock físico
        for item_data in carrito_items:
            producto = item_data['producto']
            cantidad = item_data['cantidad']

            item_venta = ItemVenta(
                cantidad=cantidad,
                precio_unitario=item_data['precio'],
                producto=producto
            )
            # Persistimos el item de la venta en la BD
            self.dao.agregar_item(nueva_venta.id, item_venta)

            self.inventario_service.registrar_salida(
                id_producto=producto.id,
                id_ubicacion=local.id,
                cantidad=cantidad,
                tipo=TipoMovimiento.VENTA,
                referencia_tipo="Venta",
                referencia_id=nueva_venta.id
            )
        # 3- Aplicamos el Patrón Strategy para procesar el pago
        medio_pago.procesar_pago(nueva_venta.total)

        pago = Pago(
            id_pago=None,
            monto=nueva_venta.total,
            referencia="Pago de Venta",
            medio_pago=medio_pago
        )
        self.dao.registrar_pago(nueva_venta.id, pago)

        # 4- Generamos el comprobante fiscal (Ticket/Factura)
        comprobante = Comprobante(
            id_comprobante=None,
            tipo="Ticket Fiscal",
            numero=f"T-{nueva_venta.id:06d}",
            fecha=datetime.now()
        )
        self.dao.guardar_comprobante(nueva_venta.id, comprobante)

        return nueva_venta

    def procesar_devolucion(self, id_venta: int, id_producto: int, cantidad: int, id_local: int):
        if cantidad <= 0:
            raise ValueError("La cantidad a devolver debe ser mayor a cero.")
        
        # 1. Registrar la devolución en la base de datos
        self.dao.registrar_devolucion(id_venta, id_producto, cantidad)
        
        # 2. Reingresar el stock al local usando tu método registrar_entrada
        self.inventario_service.registrar_entrada(
            id_producto=id_producto,
            id_ubicacion=id_local,
            cantidad=cantidad,
            tipo=TipoMovimiento.RECEPCION,  # Usamos el Enum que ya tenías importado arriba
            referencia_tipo="Devolucion de Venta",
            referencia_id=id_venta
        )

        btn_eliminar.pack(pady=5)

    def cargar_grilla(self):
        for fila in self.tree.get_children():
            self.tree.delete(fila)
        
        categorias = self.categorias_service.obtener_todas()
        for cat in categorias:
            self.tree.insert("", tk.END, values=(cat["id"], cat["nombre"]))

    def guardar(self):
        nombre = self.entry_nombre.get().strip()
        try:
            self.categorias_service.guardar_categoria(nombre)
            self.entry_nombre.delete(0, tk.END)
            self.cargar_grilla()
            messagebox.showinfo("Éxito", "Categoría guardada correctamente.")
        except Exception as e:
            messagebox.showerror("Error", str(e))

    def eliminar(self):
        seleccion = self.tree.selection()
        if not seleccion:
            messagebox.showwarning("Atención", "Seleccione una categoría para eliminar.")
            return
        
        id_cat = self.tree.item(seleccion[0])['values'][0]
        if messagebox.askyesno("Confirmar", "¿Seguro que desea eliminar esta categoría?"):
            try:
                self.categorias_service.eliminar_categoria(id_cat)
                self.cargar_grilla()
            except Exception as e:
                messagebox.showerror("Error", str(e))

            else:
                for item in faltantes:
                    tabla_faltantes.insert("", tk.END, values=(
                        item['producto_nombre'], item['stock_actual'], 
                        item['stock_minimo'], item['cantidad_sugerida']
                    ))
                lbl_estado.config(text=f"Se detectaron {len(faltantes)} productos con faltantes.", foreground="orange")
        except Exception as e:
            lbl_estado.config(text=f"Error al consultar faltantes: {e}", foreground="red")

    btn_generar = ttk.Button(frame_sugeridas, text="Calcular Faltantes", command=calcular_sugeridas)
    btn_generar.pack(pady=10)

    frame_manual = ttk.LabelFrame(ventana_compras, text="Registrar Compra (Guarda Orden y Suma Stock)")

    entry_costo = ttk.Entry(frame_manual, width=10)
    entry_costo.grid(row=1, column=3, padx=5, pady=5)

    def registrar_compra_manual():
        try:
            id_prod = int(entry_id_prod.get())
            cant = int(entry_cant.get())
            id_prov = int(entry_id_prov.get())
            costo = float(entry_costo.get())

                messagebox.showwarning("Atención", "Cantidad y costo deben ser mayores a 0", parent=ventana_compras)
                return

            # Creamos objetos genéricos en memoria para no romper nada con imports cruzados
            class DummyObj: pass
            
            prov = DummyObj()
            prov.id = id_prov
            
            prod = DummyObj()
            prod.id = id_prod
            
            item = DummyObj()
            item.producto = prod
            item.cantidad = cant
            item.precio = costo
            
            orden = DummyObj()
            orden.fecha = datetime.now()
            orden.estado = "Recibida"
            orden.proveedor = prov
            orden.total = cant * costo
            orden.items = [item]
            
            # 1. Guardar la Orden completa en la BD
            if compras_service.compras_dao:
                compras_service.compras_dao.guardar_orden(orden)
                
            # 2. Sumar el stock al Inventario

                    id_ubicacion=1,

                    referencia_id=orden.id if hasattr(orden, 'id') else 0
                )

            entry_id_prod.delete(0, tk.END)
            entry_cant.delete(0, tk.END)
            entry_id_prov.delete(0, tk.END)
            entry_costo.delete(0, tk.END)
            
            calcular_sugeridas()
            
        except ValueError:
            messagebox.showerror("Error", "Ingrese valores numéricos válidos.", parent=ventana_compras)
        except Exception as e:
            messagebox.showerror("Error de base de datos", f"Falló el guardado: {e}", parent=ventana_compras)

    btn_comprar = ttk.Button(frame_manual, text="Registrar Compra Completa", command=registrar_compra_manual)

        costo_str = self.entry_costo.get().strip()
        precio_str = self.entry_precio.get().strip()
        
        idx_categoria = self.combo_categoria.current()
        idx_proveedor = self.combo_proveedor.current()

        if not sku or not nombre or not costo_str or not precio_str or idx_categoria == -1 or idx_proveedor == -1:

        except Exception as e:
            messagebox.showerror("Error de Base de Datos", f"Ocurrió un problema: {str(e)}")

        self.entry_cuit.grid(row=1, column=1, padx=10, pady=5)

        btn_guardar = ttk.Button(marco_alta, text="Guardar Proveedor", command=self.guardar)
        btn_guardar.grid(row=2, column=0, columnspan=2, pady=10)

        marco_lista = ttk.LabelFrame(self, text="Proveedores Existentes")
        marco_lista.pack(padx=15, pady=5, fill="both", expand=True)

        self.tree = ttk.Treeview(marco_lista, columns=("ID", "Razón Social", "CUIT"), show="headings", height=8)
        self.tree.heading("ID", text="ID")
        self.tree.heading("Razón Social", text="Razón Social")
        self.tree.heading("CUIT", text="CUIT")
        self.tree.column("ID", width=40, anchor="center")
        self.tree.column("Razón Social", width=200)
        self.tree.column("CUIT", width=120, anchor="center")
        self.tree.pack(padx=10, pady=10, fill="both", expand=True)
        
        btn_eliminar = ttk.Button(marco_lista, text="Eliminar Seleccionado", command=self.eliminar)
        btn_eliminar.pack(pady=5)

    def cargar_grilla(self):
        for fila in self.tree.get_children():
            self.tree.delete(fila)
        
        proveedores = self.proveedores_service.obtener_todos()
        for prov in proveedores:
            self.tree.insert("", tk.END, values=(prov["id"], prov["razon_social"], prov["cuit"]))

    def guardar(self):
        razon = self.entry_razon.get().strip()
        cuit = self.entry_cuit.get().strip()
        try:
            self.proveedores_service.guardar_proveedor(razon, cuit)
            self.entry_razon.delete(0, tk.END)

            self.cargar_grilla()

        except Exception as e:
            messagebox.showerror("Error", str(e))

    def eliminar(self):
        seleccion = self.tree.selection()
        if not seleccion:
            messagebox.showwarning("Atención", "Seleccione un proveedor para eliminar.")
            return
        
        id_prov = self.tree.item(seleccion[0])['values'][0]
        if messagebox.askyesno("Confirmar", "¿Seguro que desea eliminar este proveedor?"):
            try:
                self.proveedores_service.eliminar_proveedor(id_prov)
                self.cargar_grilla()
            except Exception as e:
                messagebox.showerror("Error", str(e))

import tkinter as tk
from tkinter import ttk, messagebox

class VistaAltaUbicacion(tk.Toplevel):
    def __init__(self, parent, ubicacion_service):
        super().__init__(parent)
        self.title("Gestión de Ubicaciones")
        self.geometry("350x200")
        
        self.ubicacion_service = ubicacion_service
        self.crear_interfaz()

    def crear_interfaz(self):
        marco = ttk.LabelFrame(self, text="Nueva Sucursal / Depósito")
        marco.pack(padx=15, pady=15, fill="both", expand=True)

        ttk.Label(marco, text="Nombre:").grid(row=0, column=0, padx=10, pady=10, sticky="e")
        self.entry_nombre = ttk.Entry(marco)
        self.entry_nombre.grid(row=0, column=1, padx=10, pady=10)

        ttk.Label(marco, text="Dirección (Opcional):").grid(row=1, column=0, padx=10, pady=10, sticky="e")
        self.entry_direccion = ttk.Entry(marco)
        self.entry_direccion.grid(row=1, column=1, padx=10, pady=10)

        btn_guardar = ttk.Button(marco, text="Guardar Ubicación", command=self.ejecutar_guardado)
        btn_guardar.grid(row=2, column=0, columnspan=2, pady=15)

    def ejecutar_guardado(self):
        nombre = self.entry_nombre.get().strip()
        direccion = self.entry_direccion.get().strip()

        try:
            self.ubicacion_service.registrar_ubicacion(nombre, direccion)
            messagebox.showinfo("Éxito", f"La ubicación '{nombre}' fue registrada.")
            
            self.entry_nombre.delete(0, tk.END)
            self.entry_direccion.delete(0, tk.END)
            
        except ValueError as e:
            messagebox.showerror("Error de validación", str(e))
        except Exception as e:
            messagebox.showerror("Error de Base de Datos", f"Ocurrió un problema: {str(e)}")
